cmake_minimum_required(VERSION 3.22.1)
project(internotbrain CXX C)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_DIR ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp CACHE PATH "llama.cpp source")
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)        # must run on every phone, not just the build machine
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_DIR} llama)
if (ANDROID)
  add_library(internotbrain SHARED brain.cpp jni.cpp)
  target_link_libraries(internotbrain llama log)
  target_link_options(internotbrain PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(braintest brain.cpp test_main.cpp)
  target_link_libraries(braintest llama)
endif()
